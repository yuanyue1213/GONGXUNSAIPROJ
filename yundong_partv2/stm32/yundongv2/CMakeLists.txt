cmake_minimum_required(VERSION 3.22)
# Ninja 同样可能只安装在 STM32Cube 的工具目录中；保留有效的显式配置。
if(CMAKE_GENERATOR STREQUAL "Ninja" AND NOT CMAKE_MAKE_PROGRAM)
    file(GLOB _stm32_ninja_dirs "$ENV{LOCALAPPDATA}/stm32cube/bundles/ninja/*")
    list(SORT _stm32_ninja_dirs COMPARE NATURAL ORDER DESCENDING)
    find_program(CMAKE_MAKE_PROGRAM NAMES ninja
        HINTS ${_stm32_ninja_dirs} PATH_SUFFIXES bin REQUIRED)
endif()
# Select the cross compiler BEFORE project(), including plain cmake -S/-B use.
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/cmake/gcc-arm-none-eabi.cmake" CACHE FILEPATH "ARM toolchain")
endif()
project(yundongv2 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
add_executable(${PROJECT_NAME})
target_sources(${PROJECT_NAME} PRIVATE App/Src/task_display.c App/Src/lcd_bus.c)
add_subdirectory(cmake/stm32cubemx)
option(CAMERA_UART_TEST "Build the standalone UART4 camera transmit test" OFF)
option(GRIPPER_TEST "Build the standalone power-on gripper 15-degree test" OFF)
option(GRAB_TEST "Build a standalone power-on repeating raw material pick-and-store test" OFF)
option(PLACE_TEST "Build a standalone power-on repeating slot-to-station placement test" OFF)
option(NAVIGATION_ONLY "Build lidar/PC navigation without camera or arm tasks" OFF)
option(FIXED_QR "Use the configured task code instead of scanning at the QR station" OFF)
option(STACK_TEST "Build the App-started standalone visual stacking mission" OFF)
option(FORWARD_TEST "Build the App-started standalone 3000mm forward test" OFF)
option(SCREEN_TEST "Build the power-on fixed task-code LCD test" OFF)
option(TURNTABLE_TEST "Build the standalone power-on turntable zero-degree test" OFF)
option(BASE_TEST "Build the standalone power-on base angle test" OFF)
if(BASE_TEST AND (TURNTABLE_TEST OR SCREEN_TEST OR FORWARD_TEST OR STACK_TEST OR FIXED_QR OR CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(BASE_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/base_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE BASE_TEST=1)
endif()
if(TURNTABLE_TEST AND (SCREEN_TEST OR FORWARD_TEST OR STACK_TEST OR FIXED_QR OR CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(TURNTABLE_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/turntable_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE TURNTABLE_TEST=1)
endif()
if(SCREEN_TEST AND (FORWARD_TEST OR STACK_TEST OR FIXED_QR OR CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(SCREEN_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/screen_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE SCREEN_TEST=1)
endif()
if(FORWARD_TEST AND (STACK_TEST OR FIXED_QR OR CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(STACK_TEST AND (FIXED_QR OR CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(STACK_TEST)
    target_compile_definitions(${PROJECT_NAME} PRIVATE STACK_TEST=1 FIXED_QR=1)
endif()
if(FIXED_QR AND (CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR PLACE_TEST OR NAVIGATION_ONLY))
    message(FATAL_ERROR "Fixed QR firmware requires the full mission")
endif()
if(FIXED_QR)
    target_compile_definitions(${PROJECT_NAME} PRIVATE FIXED_QR=1)
endif()
if((CAMERA_UART_TEST AND GRIPPER_TEST) OR (NAVIGATION_ONLY AND (CAMERA_UART_TEST OR GRIPPER_TEST))
   OR (GRAB_TEST AND (CAMERA_UART_TEST OR GRIPPER_TEST OR NAVIGATION_ONLY))
   OR (PLACE_TEST AND (CAMERA_UART_TEST OR GRIPPER_TEST OR GRAB_TEST OR NAVIGATION_ONLY)))
    message(FATAL_ERROR "Select only one standalone test firmware")
endif()
if(CAMERA_UART_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/camera_uart_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE CAMERA_UART_TEST=1)
endif()
if(GRIPPER_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/gripper_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE GRIPPER_TEST=1)
endif()
if(GRAB_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/grab_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE GRAB_TEST=1)
endif()
if(PLACE_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/place_test.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE PLACE_TEST=1)
endif()
if(NOT SCREEN_TEST)
target_sources(${PROJECT_NAME} PRIVATE
    App/Src/zdt_motor.c App/Src/zdt_motor_uart.c
    App/Src/imu_link.c App/Src/start_link.c App/Src/motion_control.c
    App/Src/route_plan.c
)
if(FORWARD_TEST)
    target_sources(${PROJECT_NAME} PRIVATE Tests/forward_test.c Tests/straight_motion.c)
    # 复用底盘构建条件，不编译视觉和机械臂业务模块。
    target_compile_definitions(${PROJECT_NAME} PRIVATE FORWARD_TEST=1 NAVIGATION_ONLY=1)
elseif(NAVIGATION_ONLY)
    target_sources(${PROJECT_NAME} PRIVATE Tests/navigation_only.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE NAVIGATION_ONLY=1)
else()
    target_sources(${PROJECT_NAME} PRIVATE
        App/Src/lift_motor.c App/Src/servo_control.c App/Src/servo_trajectory.c
        App/Src/camera_link.c App/Src/mission_runner.c App/Src/flow_check.c App/Src/arm_flow.c)
endif()
endif() # NOT SCREEN_TEST
target_include_directories(${PROJECT_NAME} PRIVATE App/Inc ../../shared)
target_link_libraries(${PROJECT_NAME} stm32cubemx m)
# Camera requests contain floating-point coordinates; nano needs explicit printf support.
if(NOT NAVIGATION_ONLY AND NOT FORWARD_TEST)
    target_link_options(${PROJECT_NAME} PRIVATE "LINKER:-u,_printf_float")
endif()
set_property(TARGET ${PROJECT_NAME} APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/STM32H743xx_FLASH.ld")
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.bin
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}>
    VERBATIM)
